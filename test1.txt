; PROGRAM 2: SEARCH WORD IN PARAGRAPH
; LOC 0

STX 1,0        ; 32 → X1 = 1000 (paragraph base)

; ==========================
; 2) Initialize period & counters
; ==========================
AIR 0,46       ; 33 → period ASCII
STR 0,0,31     ; 34 → memory[31] = 46 (period)
AIR 0,0        ; 35 → word number counter
STR 0,0,20     ; 36 → memory[20] = 0
AIR 0,1        ; 37 → sentence number start 1
STR 0,0,21     ; 38 → memory[21] = 1

; ==========================
; 3) Input search word (10 chars) → memory 0–9
; ==========================
IN 0,0         ; 39
STR 0,0,0      ; 40
IN 0,0
STR 0,0,1
IN 0,0

IN 0,0

IN 0,0

; ==========================
; 4) Initialize X2 = jump table base (50)
; ==========================
LDA 2,0,50     ; 50 → X2 = 50
STX 2,0        ; 51

; ==========================
; 5) Unrolled 60 word blocks
; ==========================
; Each block: load 10 chars → compare → store word & sentence if match → increment X1 → check period → jump via X2
; JNE using X2 for absolute addresses >31
; Structure repeated for all 60 blocks (illustrated for first block)

; ---- BLOCK 1 ----
LDR 0,1,0
STR 0,0,10
LDR 0,1,1
STR 0,0,11
LDR 0,1,2

LDR 0,1,8
STR 0,0,18
LDR 0,1,9
STR 0,0,19

; Compare with search word
LDR 0,0,10
LDR 1,0,0
SMR 0,0,0
JNE 0,2,1     ; via X2 → next block (offset 1)
LDR 0,0,11
LDR 1,0,1
SMR 0,0,1
JNE 0,2,1
LDR 0,0,12
LDR 1,0,2
SMR 0,0,2

LDR 0,0,13
LDR 1,0,3
SMR 0,0,3
JNE 0,2,1
LDR 0,0,14

JNE 0,2,1
LDR 0,0,15

JNE 0,2,1
LDR 0,0,18
LDR 1,0,8
SMR 0,0,8
JNE 0,2,1
LDR 0,0,19
LDR 1,0,9
SMR 0,0,9
JNE 0,2,1

; Match → store word & sentence
AIR 0,1
STR 0,0,20
AIR 0,0
STR 0,0,21

; Increment paragraph pointer (10 chars)
AIR 0,10
LDA 0,1,0
STX 1,0

; Check period
LDR 0,0,19
SMR 0,0,31
JNE 0,2,2     ; not period → skip sentence increment
LDR 0,0,21
AIR 0,1
STR 0,0,21

; ---- Next blocks: BLOCK 2 → BLOCK 60 ----
; Repeat same structure, increment X1 for each, adjust JNE via X2
; (Each JNE offset ≤ 31, X2 points to next block)

; ==========================
; 6) OUTPUT WORD & SENTENCE NUMBER
; ==========================
LDR 0,0,20
OUT 0,1
LDR 0,0,21
OUT 0,1

; ==========================
; 7) HLT
; ==========================
HLT
